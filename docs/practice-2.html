def my_sum(*args:list|tuple, **kwargs) -> int|float:
    точность = kwargs.get("точность",5)   
    s = 0
    for a in args:
        if isinstance(a, (list, tuple)):
            s += my_sum(*a, **kwargs)
        else:
            s += a 
    s = round(s, точность)
    return s

    '''
    : 1,1,2,3,5,8,13,21 # Фибоначчи 
    : f(n) = f(n-2) + f(n-1)
    '''
    return 1 if n == 1 or n == 2 else fib(n-2) + fib(n-1)

class Point:    
    def __init__(self, x = 0, y = 0):
        self.x = x
        self.y = y
        
    def __str__(self):
        return f'Я точка ({self.x}, {self.y})'
    
    def __add__(self, other):
        return Point(self.x + other.x,self.y + other.y)
