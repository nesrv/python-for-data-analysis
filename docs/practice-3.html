RETURNS integer
RETURN CASE 
	WHEN a > b THEN maximum(a,c)
	ELSE maximum (b,c)
END;

CREATE or RAPLACE FUNCTION maximum(
    a anycompatible, 
    b anycompatible, 
    c anycompatible DEFAULT NULL
) RETURNS anycompatible 
AS $$
SELECT CASE
         WHEN c IS NULL THEN
             x
         ELSE
             CASE WHEN x > c THEN x ELSE c END
       END
FROM (
    SELECT CASE WHEN a > b THEN a ELSE b END
) max2(x);
$$ LANGUAGE sql;


SELECT maximum(10, 300.14, 30E3), maximum ('a', 'v');

CREATE FUNCTION fmt (IN phone text, OUT code text, OUT num text)
AS $$
BEGIN
    IF phone ~ '^[0-9]*$' AND length(phone) = 10 THEN
        code := substr(phone,1,3);
        num  := substr(phone,4);
    ELSE
        code := NULL;
        num  := NULL;
    END IF;
END
$$ LANGUAGE plpgsql;

SELECT fmt('8122128506');

DO $$
DECLARE
    code text := (fmt('4952128506')).code;
begin
	CASE code
        WHEN '495','499' THEN
            RAISE NOTICE '% — Москва', code;
        WHEN '812' THEN
            RAISE NOTICE '% — Санкт-Петербург', code;
        WHEN '384' THEN
            RAISE NOTICE '% — Кемеровская область', code;
        ELSE
            RAISE NOTICE '% — Прочие', code;
    END CASE;
END
$$;
end
